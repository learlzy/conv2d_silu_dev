- 编译计算图：
atc --framework=5 --model=yolo26s.onnx \
--input_format=NCHW --input_shape="images:8,3,640,640" \
--soc_version=Ascend910B3  --output=yolo26s

- 输出完整计算图信息：
atc --mode=1 \
--om=yolo26s.om \
--json=yolo26s_om.json

- 输出所有阶段的计算图：
cd build
export DUMP_GE_GRAPH=1
atc --framework=5 --model=../yolov8s_subgraph.onnx \
--input_format=NCHW --input_shape="images:8,3,640,640" \
--soc_version=Ascend910B3  --output=yolov8s_subgraph
